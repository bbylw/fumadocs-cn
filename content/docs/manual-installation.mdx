您也可以让 [Fumadocs CLI](/docs/cli#init) 自动为您完成设定：

```npm
npx @fumadocs/cli init
```
继续之前，请确认您已设定好：

## 开始使用 [#getting-started]

安装套件：


  ```npm
  npm i fumadocs-core fumadocs-ui@npm:@fumadocs/base-ui
  ```



  ```npm
  npm i fumadocs-core fumadocs-ui@npm:@fumadocs/base-ui @fumadocs/astro @astrojs/react react react-dom @types/mdx
  ```


与范本相同，此指令会将 `fumadocs-ui` 安装为 `@fumadocs/base-ui` 的别名，也就是 Fumadocs UI 的 [Base UI](/docs/ui/component-library) 版本。若要改用 Radix UI，请直接安装 `fumadocs-ui`，不要设定别名。

### 样式 [#styles]

请将以下内容加入您的 Tailwind CSS 档案：

```css
@import 'tailwindcss';
/* [!code ++:2] */
@import 'fumadocs-ui/css/neutral.css';
@import 'fumadocs-ui/css/preset.css';
```


  ```tsx title="app/layout.tsx"
  import { RootProvider } from 'fumadocs-ui/provider/next';
  import type { ReactNode } from 'react';

  export default function Layout({ children }: { children: ReactNode }) {
    return (
      <html lang="en" suppressHydrationWarning>
        <body
          // 必要的样式 [!code ++:2]
          className="flex flex-col min-h-screen"
        >
          <RootProvider>{children}</RootProvider>
        </body>
      </html>
    );
  }
  ```



  ```tsx title="app/root.tsx"
  import { Links, Meta, Scripts, ScrollRestoration } from 'react-router';
  import { RootProvider } from 'fumadocs-ui/provider/react-router';
  import './app.css';

  export function Layout({ children }: { children: React.ReactNode }) {
    return (
      <html lang="en" suppressHydrationWarning>
        <head>
          <meta charSet="utf-8" />
          <meta name="viewport" content="width=device-width, initial-scale=1" />
          <Meta />
          <Links />
        </head>
        <body className="flex flex-col min-h-screen">
          {/* [!code ++] */}
          <RootProvider>{children}</RootProvider>
          <ScrollRestoration />
          <Scripts />
        </body>
      </html>
    );
  }
  ```



  ```tsx title="src/routes/__root.tsx"
  import { createRootRoute, HeadContent, Outlet, Scripts } from '@tanstack/react-router';
  import * as React from 'react';
  import { RootProvider } from 'fumadocs-ui/provider/tanstack';

  export const Route = createRootRoute({
    component: RootComponent,
  });

  function RootComponent() {
    return (
      <RootDocument>
        <Outlet />
      </RootDocument>
    );
  }

  function RootDocument({ children }: { children: React.ReactNode }) {
    return (
      <html suppressHydrationWarning>
        <head>
          <HeadContent />
        </head>
        <body className="flex flex-col min-h-screen">
          {/* [!code ++] */}
          <RootProvider>{children}</RootProvider>
          <Scripts />
        </body>
      </html>
    );
  }
  ```



  请在用户端元件中，用 [Root Provider](/docs/ui/layouts/root-provider) 包住整个应用程式：

  


```tsx
'use client';
import type { ReactNode } from 'react';
import { RootProvider } from 'fumadocs-ui/provider/waku';

export function Provider({ children }: { children: ReactNode }) {
  return <RootProvider>{children}</RootProvider>;
}
```

  ### 整合 [#integration]

  将 Fumadocs 整合加入您的 Astro 设定：

  ```js title="astro.config.mjs"
  import { defineConfig } from 'astro/config';
  import tailwindcss from '@tailwindcss/vite';
  import fumadocs from '@fumadocs/astro';

  export default defineConfig({
    integrations: [fumadocs({ components: './src/components/mdx.tsx' })],
    vite: {
      plugins: [tailwindcss()],
    },
  });
  ```
定义您的内容集合：

  


```ts title="src/content.config.ts"
import { defineCollection } from 'astro:content';
import { metaLoader, pageLoader } from '@fumadocs/astro/loaders';

export const collections = {
  docs: defineCollection({ loader: pageLoader() }),
  meta: defineCollection({ loader: metaLoader() }),
};
```




  根据您的集合建立 Fumadocs 来源：

  


```ts title="src/lib/source.ts"
import { getCollection } from 'astro:content';
import { loader } from 'fumadocs-core/source';
import { toFumadocsSource } from '@fumadocs/astro/source';

export const source = loader({
  source: toFumadocsSource(await getCollection('docs'), await getCollection('meta')),
  baseUrl: '/docs',
});
```




  若要在同一个 `loader()` 中与其他来源合并，请将 `{ baseDir }` 作为 `toFumadocsSource()` 的第三个引数传入，让其档案放在指定目录下。


### 路由 [#routes]


  请更新您的路由：

  ```ts title="app/routes.ts"
  import { type RouteConfig, index, route } from '@react-router/dev/routes';

  export default [
    index('routes/home.tsx'),
    // [!code ++:2]
    route('docs/*', 'routes/docs.tsx'),
    route('api/search', 'routes/search.ts'),
  ] satisfies RouteConfig;
  ```

  请建立以下档案：

  


```tsx title="lib/layout.shared.tsx"
import type { BaseLayoutProps } from 'fumadocs-ui/layouts/shared';

export function baseOptions(): BaseLayoutProps {
  return {
    nav: {
      title: 'My App',
    },
  };
}
```




  


```tsx
import defaultMdxComponents from 'fumadocs-ui/mdx';
import type { MDXComponents } from 'mdx/types';

export function getMDXComponents(components?: MDXComponents) {
  return {
    ...defaultMdxComponents,
    ...components,
  } satisfies MDXComponents;
}

export const useMDXComponents = getMDXComponents;

declare global {
  type MDXProvidedComponents = ReturnType<typeof getMDXComponents>;
}
```




  


```tsx
import { source } from '@/lib/source';
import { DocsLayout } from 'fumadocs-ui/layouts/docs';
import { baseOptions } from '@/lib/layout.shared';

export default function Layout({ children }: LayoutProps<'/docs'>) {
  return (
    <DocsLayout tree={source.getPageTree()} {...baseOptions()}>
      {children}
    </DocsLayout>
  );
}
```




  


```tsx
import { source } from '@/lib/source';
import { DocsBody, DocsDescription, DocsPage, DocsTitle } from 'fumadocs-ui/layouts/docs/page';
import { notFound } from 'next/navigation';
import { getMDXComponents } from '@/components/mdx';
import type { Metadata } from 'next';
import { createRelativeLink } from 'fumadocs-ui/mdx';

export default async function Page(props: PageProps<'/docs/[[...slug]]'>) {
  const params = await props.params;
  const page = source.getPage(params.slug);
  if (!page) notFound();

  const MDX = page.data.body;

  return (
    <DocsPage toc={page.data.toc} full={page.data.full}>
      <DocsTitle>{page.data.title}</DocsTitle>
      <DocsDescription>{page.data.description}</DocsDescription>
      <DocsBody>
        <MDX
          components={getMDXComponents({
            // this allows you to link to other pages with relative file paths
            a: createRelativeLink(source, page),
          })}
        />
      </DocsBody>
    </DocsPage>
  );
}

export async function generateStaticParams() {
  return source.generateParams();
}

export async function generateMetadata(props: PageProps<'/docs/[[...slug]]'>): Promise<Metadata> {
  const params = await props.params;
  const page = source.getPage(params.slug);
  if (!page) notFound();

  return {
    title: page.data.title,
    description: page.data.description,
  };
}
```




  在 [RSC 模式](https://reactrouter.com/how-to/react-server-components)下，docs 路由汇出的是 `ServerComponent`，而不是载入器与页面元件：

  


```ts
import { source } from '@/lib/source';
import { createFromSource } from 'fumadocs-core/search/server';

export const { GET } = createFromSource(source, {
  // https://docs.orama.com/docs/orama-js/supported-languages
  language: 'english',
});
```





搜寻路由使用 ZBSearch，设定方式请参阅[文件搜寻](/docs/headless/search)。

### 完成 [#finish]

启动开发伺服器，即可在 `content/docs` 中撰写您的文件：

```mdx title="content/docs/index.mdx"
---
title: Hello World
---

## Introduction

I love Fumadocs.
```

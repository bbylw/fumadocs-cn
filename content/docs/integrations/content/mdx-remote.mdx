title: MDX Remote

---

Fumadocs 提供 **MDX Remote** 套件，可协助您将 Markdown/MDX 内容来源整合至 Fumadocs。
您可以把它想成内建 Fumadocs 外挂的 `next-mdx-remote`。

<Callout type="warn">
  请勿将不受信任的 MDX 传给编译器，因为它会在您的应用程式中以 JavaScript 执行。请改将不受信任的内容以 Markdown 格式编译，这样它就无法执行 JavaScript 或原始 HTML：

  ```ts
  import { createCompiler } from '@fumadocs/mdx-remote';

  const compiler = createCompiler({
    format: 'md',
    // 略过为取得图片尺寸而读取档案及撷取 URL
    remarkImageOptions: false,
  });
  ```
</Callout>

## 设定 [#setup]

```package-install
@fumadocs/mdx-remote
```

它的主要功能是 MDX 编译器，可将 MDX 内容编译为 JSX 节点。
由于它不使用打包工具，因此有以下限制：

* 图片必须在执行阶段进行最佳化。
* MDX 档案中无法使用汇入 (import) 与汇出 (export) 。

### 使用方式 [#usage]

建立编译器：

```ts title="lib/compiler.ts"
import { createCompiler } from '@fumadocs/mdx-remote';

export const compiler = createCompiler({
  // 选项
});
```

编译页面内容，其中 `@/lib/source` 代表您的内容来源。


  ```tsx title="app/docs/[[...slug]]/page.tsx"
  import { DocsBody, DocsPage } from 'fumadocs-ui/layouts/docs/page';
  import { compiler } from '@/lib/compiler';
  import { getPage } from '@/lib/source';
  import { getMDXComponents } from '@/components/mdx';

  export default async function Page(props: PageProps<'/docs/[[...slug]]'>) {
    const { slug } = await props.params;
    const page = getPage(slug);
    const { body: MdxContent, toc } = await compiler.compile({
      source: page.content,
    });

    return (
      <DocsPage toc={toc}>
        <DocsBody>
          <MdxContent components={getMDXComponents()} />
        </DocsBody>
      </DocsPage>
    );
  }
  ```



  ```tsx title="app/routes/docs.tsx"
  import type { Route } from './+types/docs';
  import { useMemo } from 'react';
  import { executeMdxSync } from '@fumadocs/mdx-remote/client';
  import { useFumadocsLoader } from 'fumadocs-core/source/client';
  import { DocsLayout } from 'fumadocs-ui/layouts/docs';
  import { DocsBody, DocsPage } from 'fumadocs-ui/layouts/docs/page';
  import { compiler } from '@/lib/compiler';
  import { getPage, source } from '@/lib/source';
  import { getMDXComponents } from '@/components/mdx';

  export async function loader({ params }: Route.LoaderArgs) {
    const page = getPage(params['*'].split('/').filter((v) => v.length > 0));

    return {
      // 编译为 JavaScript，并在用户端执行
      compiled: String(await compiler.compileFile(page.content)),
      pageTree: await source.serializePageTree(source.getPageTree()),
    };
  }

  export default function Page({ loaderData }: Route.ComponentProps) {
    const { compiled, pageTree } = useFumadocsLoader(loaderData);
    const { default: MdxContent, toc } = useMemo(() => executeMdxSync(compiled), [compiled]);

    return (
      <DocsLayout tree={pageTree}>
        <DocsPage toc={toc}>
          <DocsBody>
            <MdxContent components={getMDXComponents()} />
          </DocsBody>
        </DocsPage>
      </DocsLayout>
    );
  }
  ```



  ```tsx title="src/routes/docs/$.tsx"
  import { createFileRoute } from '@tanstack/react-router';
  import { createServerFn } from '@tanstack/react-start';
  import { useMemo } from 'react';
  import { executeMdxSync } from '@fumadocs/mdx-remote/client';
  import { useFumadocsLoader } from 'fumadocs-core/source/client';
  import { DocsLayout } from 'fumadocs-ui/layouts/docs';
  import { DocsBody, DocsPage } from 'fumadocs-ui/layouts/docs/page';
  import { compiler } from '@/lib/compiler';
  import { getPage, source } from '@/lib/source';
  import { getMDXComponents } from '@/components/mdx';

  export const Route = createFileRoute('/docs/$')({
    component: Page,
    loader: ({ params }) => serverLoader({ data: params._splat?.split('/') ?? [] }),
  });

  const serverLoader = createServerFn({
    method: 'GET',
  })
    .validator((slugs: string[]) => slugs)
    .handler(async ({ data: slugs }) => {
      const page = getPage(slugs);

      return {
        // 编译成 JavaScript，于用户端执行
        compiled: String(await compiler.compileFile(page.content)),
        pageTree: await source.serializePageTree(source.getPageTree()),
      };
    });

  function Page() {
    const { compiled, pageTree } = useFumadocsLoader(Route.useLoaderData());
    const { default: MdxContent, toc } = useMemo(() => executeMdxSync(compiled), [compiled]);

    return (
      <DocsLayout tree={pageTree}>
        <DocsPage toc={toc}>
          <DocsBody>
            <MdxContent components={getMDXComponents()} />
          </DocsBody>
        </DocsPage>
      </DocsLayout>
    );
  }
  ```

  ```astro tab="src/pages/docs/[...slug].astro"
  ---
  import type { GetStaticPaths, InferGetStaticPropsType } from 'astro';
  import Layout from '@/layouts/Layout.astro';
  import { Docs } from '@/components/docs';
  import { compiler } from '@/lib/compiler';
  import { getPage, source } from '@/lib/source';

  export const prerender = true;

  export const getStaticPaths = (() => {
    return source.getPages().map((page) => ({
      params: { slug: decodeURI(page.slugs.join('/')) || undefined },
      props: { page },
    }));
  }) satisfies GetStaticPaths;

  type Props = InferGetStaticPropsType<typeof getStaticPaths>;

  const { page } = Astro.props;
  const pageTree = await source.serializePageTree(source.getPageTree());
  // 编译为 JavaScript，于 island 中执行
  const compiled = String(await compiler.compileFile(getPage(page.slugs).content));
  ---

  <Layout title={page.data.title}>
    <Docs
      client:load
      pathname={Astro.url.pathname}
      params={Astro.params}
      tree={pageTree}
      compiled={compiled}
    />
  </Layout>
  ```

  ```tsx tab="src/components/docs.tsx"
  import { useMemo } from 'react';
  import { executeMdxSync } from '@fumadocs/mdx-remote/client';
  import { type SerializedPageTree, useFumadocsLoader } from 'fumadocs-core/source/client';
  import { DocsLayout } from 'fumadocs-ui/layouts/docs';
  import { DocsBody, DocsPage } from 'fumadocs-ui/layouts/docs/page';
  import { Provider, type ProviderProps } from '@/components/provider';
  import { getMDXComponents } from '@/components/mdx';

  export interface DocsProps extends ProviderProps {
    tree: SerializedPageTree;
    compiled: string;
  }

  export function Docs({ tree, compiled, ...props }: DocsProps) {
    const data = useFumadocsLoader({ tree });
    const { default: MdxContent, toc } = useMemo(() => executeMdxSync(compiled), [compiled]);

    return (
      <Provider {...props}>
        <DocsLayout tree={data.tree}>
          <DocsPage toc={toc}>
            <DocsBody>
              <MdxContent components={getMDXComponents()} />
            </DocsBody>
          </DocsPage>
        </DocsLayout>
      </Provider>
    );
  }
  ```


> `compile()` 回传的 `frontmatter` 属性不会经过验证，请使用 `zod` 等函式库自行验证。

### 图片 [#images]

在 Vercel 等无伺服器平台上，原本的 `public` 资料夹 (包含图片等静态资源) 会在正式环境建置完成后被移除，
因此 MDX 编译器可能无法再存取 `public` 中的本机图片。

引用图片时，请务必使用 URL。

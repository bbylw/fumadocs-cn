## 概览 [#overview]

若你使用的 React 框架支持中间件，可以通过 Fumadocs 的协商 API 实现基本功能。
在 Waku 上，请在 `src/middleware` 的 [Hono 中间件](https://hono.dev/docs/guides/middleware)中使用它。
在 Astro 上，请在[中间件](https://docs.astro.build/en/guides/middleware)中使用它；此中间件只会在依请求渲染的页面上运行。

### 接受 Markdown [#accept-markdown]

依据 `Accept` 标头返回 Markdown 或 HTML，可改善 AI 代理的使用体验。

```ts tab="Next.js" title="proxy.ts"
import { NextRequest, NextResponse } from 'next/server';
import { isMarkdownPreferred, rewritePath } from 'fumadocs-core/negotiation';

const { rewrite: rewriteLLM } = rewritePath('/docs/*path', '/llms.mdx/*path');

export default function proxy(request: NextRequest) {
  if (isMarkdownPreferred(request)) {
    const result = rewriteLLM(request.nextUrl.pathname);

    if (result) {
      return NextResponse.rewrite(new URL(result, request.nextUrl));
    }
  }

  return NextResponse.next();
}
```

```tsx tab="React Router" title="app/root.tsx"
import type { Route } from './+types/root';
import { isMarkdownPreferred, rewritePath } from 'fumadocs-core/negotiation';

const { rewrite: rewriteLLM } = rewritePath('/docs/*path', '/llms.mdx/*path');

const serverMiddleware: Route.MiddlewareFunction = async ({ request }, next) => {
  const url = new URL(request.url);
  const result = isMarkdownPreferred(request) && rewriteLLM(url.pathname);

  // `Response.redirect()` 的 headers 无法修改，因此直接创建回应
  if (result)
    return new Response(null, {
      status: 302,
      headers: { Location: new URL(result, url).toString(), Vary: 'Accept' },
    });

  return next();
};

export const middleware = [serverMiddleware];
```

```ts tab="TanStack Start" title="src/start.ts"
import { createMiddleware, createStart } from '@tanstack/react-start';
import { redirect } from '@tanstack/react-router';
import { isMarkdownPreferred, rewritePath } from 'fumadocs-core/negotiation';

const { rewrite: rewriteLLM } = rewritePath('/docs/*path', '/llms.mdx/*path');

const llmMiddleware = createMiddleware().server(({ next, request }) => {
  const url = new URL(request.url);
  const result = isMarkdownPreferred(request) && rewriteLLM(url.pathname);

  // 重定向到 Markdown URL
  if (result) throw redirect({ href: new URL(result, url).href, headers: { Vary: 'Accept' } });

  return next();
});

export const startInstance = createStart(() => ({
  requestMiddleware: [llmMiddleware],
}));
```
